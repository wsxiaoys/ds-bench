import altair as alt
from vega_datasets import data

brush = alt.selection_interval(name='brush')

bar = alt.Chart(data.movies.url).mark_bar().encode(
    x='count:Q',
    y=alt.Y('Major_Genre:N', sort='-x')
).transform_filter(
    (alt.datum.IMDB_Rating != None) & (alt.datum.Rotten_Tomatoes_Rating != None)
).transform_filter(
    brush
).transform_aggregate(
    count='count()',
    groupby=['Major_Genre']
).transform_window(
    rank='rank()',
    sort=[alt.SortField('count', order='descending')]
).transform_filter(
    alt.datum.rank <= 8
)

print(bar.to_dict())

import altair as alt
from vega_datasets import data

brush = alt.selection_interval(name='brush')

scatter = alt.Chart(data.movies.url).mark_point().encode(
    x='IMDB_Rating:Q',
    y='Rotten_Tomatoes_Rating:Q',
    color=alt.when(brush).then('Major_Genre:N').otherwise(alt.value('lightgray'))
).add_params(
    brush
).transform_filter(
    (alt.datum.IMDB_Rating != None) & (alt.datum.Rotten_Tomatoes_Rating != None)
)

bar = alt.Chart(data.movies.url).mark_bar().encode(
    x='count:Q',
    y=alt.Y('Major_Genre:N', sort='-x')
).transform_filter(
    (alt.datum.IMDB_Rating != None) & (alt.datum.Rotten_Tomatoes_Rating != None)
).transform_filter(
    brush
).transform_aggregate(
    count='count()',
    groupby=['Major_Genre']
).transform_window(
    rank='rank()',
    sort=[alt.SortField('count', order='descending')]
).transform_filter(
    alt.datum.rank <= 8
)

hist = alt.Chart(data.movies.url).mark_bar().encode(
    x=alt.X('IMDB_Rating:Q', bin=alt.Bin(maxbins=20)),
    y='count():Q'
).transform_filter(
    (alt.datum.IMDB_Rating != None) & (alt.datum.Rotten_Tomatoes_Rating != None)
).transform_filter(
    brush
)

dashboard = scatter & (bar | hist)
print(dashboard.to_dict().keys())
