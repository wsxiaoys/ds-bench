files = glob.glob(pattern)

all_values = []
worker_parts = {}

for f in files:
    filename = os.path.basename(f)
    parts = filename.split('-')
    worker_idx = int(parts[3])
    part_num = int(parts[5].split('.')[0])
    
    worker_parts.setdefault(worker_idx, []).append((part_num, f))
    
    with open(f, 'r') as fh:
        lines = fh.readlines()
        print(f'{filename}: {len(lines)} lines')
        for line in lines:
            data = json.loads(line)
            assert 'worker' in data
            assert 'value' in data
            assert data['worker'] == worker_idx
            all_values.append(data['value'])

print('Total records:', len(all_values))
