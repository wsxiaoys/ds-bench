import nltk
from nltk.corpus import wordnet as wn

def expand_query(query: str) -> list[str]:
    query = query.lower()
    tokens = query.split()
    expanded_terms = []
    seen_all = set()
    for token in tokens:
        if token not in seen_all:
            seen_all.add(token)
            expanded_terms.append(token)
        synonyms_for_token = []
        seen_syns = set()
        for synset in wn.synsets(token):
            for lemma in synset.lemmas():
                name = lemma.name().lower()
                if '_' in name or ' ' in name:
                    continue
                if name == token:
                    continue
                if name not in seen_syns:
                    seen_syns.add(name)
                    synonyms_for_token.append(name)
                    if len(synonyms_for_token) == 3:
                        break
            if len(synonyms_for_token) == 3:
                break
        for syn in synonyms_for_token:
            if syn not in seen_all:
                seen_all.add(syn)
                expanded_terms.append(syn)
    return expanded_terms

print('car:', expand_query('car'))
print('automobile:', expand_query('automobile'))
print('fast car:', expand_query('fast car'))
