import pickle

class UnionFind:
    def __init__(self):
        self.parent = {}
        self.size = {}

    def find(self, i):
        if i not in self.parent:
            self.parent[i] = i
            self.size[i] = 1
            return i
        curr = i
        path = []
        while self.parent[curr] != curr:
            path.append(curr)
            curr = self.parent[curr]
        for node in path:
            self.parent[node] = curr
        return curr

    def union(self, i, j):
        root_i = self.find(i)
        root_j = self.find(j)
        if root_i == root_j:
            return False, self.size[root_i]
        if self.size[root_i] < self.size[root_j]:
            root_i, root_j = root_j, root_i
        self.parent[root_j] = root_i
        self.size[root_i] += self.size[root_j]
        return True, self.size[root_i]

uf = UnionFind()
uf.union('A', 'B')
data = pickle.dumps(uf)
uf2 = pickle.loads(data)
print('Pickle check:', uf2.find('A') == uf2.find('B'), uf2.size[uf2.find('A')])
