import altair as alt
from vega_datasets import data

barley_data = data.barley.url

bar_chart = alt.Chart(barley_data).mark_bar().encode(
    x=alt.X('site:N').sort(field='yield', op='mean', order='descending'),
    y=alt.Y('mean(yield):Q'),
    xOffset=alt.XOffset('variety:N'),
    color=alt.Color('variety:N').scale(scheme='tableau10')
)

tick_chart = alt.Chart(barley_data).mark_tick(
    color='black',
    size=25,
    thickness=3
).transform_aggregate(
    mean_yield='mean(yield)',
    groupby=['site', 'year']
).encode(
    x=alt.X('site:N').sort(field='mean_yield', op='mean', order='descending'),
    y=alt.Y('mean_yield:Q')
)

chart = alt.layer(bar_chart, tick_chart).facet(
    facet=alt.Facet('year:O')
)

print(chart.to_dict().keys())
