import json, re, pathlib

html = pathlib.Path("streamgraph.html").read_text()

# Extract the JSON spec embedded by Vega-Embed
# Altair writes:  vegaEmbed('#vis', <SPEC>, ...)
m = re.search(r'vegaEmbed\([^,]+,\s*(\{.*\}),\s*\{', html, re.DOTALL)
if not m:
    # fallback: grab the first {...} block after vegaEmbed
    m = re.search(r'vegaEmbed\(.*?,\s*(\{.+)', html, re.DOTALL)

# Safer: find the spec dict passed to vegaEmbed
# Altair ≥ 5 embeds as:  var spec = {...};  vegaEmbed('#vis', spec, ...)
spec_match = re.search(r'var spec\s*=\s*(\{.*?\});\s*var embedOpt', html, re.DOTALL)
if spec_match:
    spec = json.loads(spec_match.group(1))
else:
    # Try direct extraction
    idx = html.find('vegaEmbed(')
    sub = html[idx:]
    # find first { after the element selector argument
    brace_start = sub.index('{')
    depth = 0
    for i, ch in enumerate(sub[brace_start:], brace_start):
        if ch == '{': depth += 1
        elif ch == '}':

                break
