import altair as alt, pandas as pd, json

# Reproduce the spec and inspect it directly (no HTML parsing needed)
df = pd.read_csv("data/measurements.csv")
features = ["temperature", "pressure", "humidity", "vibration"]

brush = alt.selection_interval(name="brush", resolve="global", empty=True)

base = (
    alt.Chart(df)
    .mark_point(filled=True, size=40, opacity=0.7)
    .encode(
        x=alt.X(alt.repeat("column"), type="quantitative"),
        y=alt.Y(alt.repeat("row"),    type="quantitative"),
        color=alt.condition(
            brush,
            alt.Color("machine_class:N", legend=alt.Legend(title="Machine Class")),
            alt.value("lightgray"),
        ),
        opacity=alt.condition(brush, alt.value(0.85), alt.value(0.25)),
    )
    .add_params(brush)
    .properties(width=160, height=160)
)

splom = base.repeat(row=features, column=features)

spec = json.loads(splom.to_json())

# 1. Top-level repeat
repeat = spec.get("repeat", {})
print("repeat.row   :", repeat.get("row"))
