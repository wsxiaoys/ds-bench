    tokens = re.findall(r'[a-z0-9]+', text.lower())

import lancedb
import numpy as np
import hashlib
import re

def embed_query(text):
    tokens = re.findall(r'[a-z0-9]+', text.lower())
    vec = np.zeros(32, dtype=np.float32)
    for tok in tokens:
        idx = int(hashlib.md5(tok.encode('utf-8')).hexdigest(), 16) % 32
        vec[idx] += 1.0
    norm = np.linalg.norm(vec)
    if norm > 0:
        vec = vec / norm
    return vec.astype(np.float32)
