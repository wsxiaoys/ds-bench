```tsx
import { render, route } from "rwsdk/router";
import { defineApp } from "rwsdk/worker";

import { Document } from "@/app/document";
import { setCommonHeaders } from "@/app/headers";
import { Home } from "@/app/pages/home";

export type AppContext = {};

export default defineApp([
  setCommonHeaders(),
  ({ ctx }) => {
    // setup ctx here
    ctx;
  },
  render(Document, [
    route("/", Home),
    route("/ping", () => <h1>Pong!</h1>),
    route("/about", () => (
      <>
        <h1>About RedwoodSDK</h1>
        <p>React framework for Cloudflare</p>
      </>
    )),
    route("/greet/:name", ({ params }) => <h1>Hello, {params.name}!</h1>),
  ]),
  route("/status", () =>
    new Response(JSON.stringify({ ok: true, name: "rwsdk" }), {
      headers: { "content-type": "application/json" },
    })
  ),
]);
```

- **`GET /ping`** — Lives under `render(Document, ...)`. Returns JSX `<h1>Pong!</h1>` which renders to full HTML via the `Document` wrapper.
- **`GET /about`** — Lives under `render(Document, ...)`. Returns JSX with an `<h1>About RedwoodSDK</h1>` heading and a `<p>React framework for Cloudflare</p>` paragraph.
- **`GET /greet/:name`** — Lives under `render(Document, ...)`. Uses the `params` argument to read the `:name` path parameter and renders `Hello, {params.name}!` inside an `<h1>`.
- **`GET /status`** — Lives **outside** the `render(Document, ...)` block, since it returns a raw `Response` object (not JSX) and does not need the document wrapper.
