import json, os, urllib.request

run_id = open('/logs/artifacts/run-id').read().strip()

headers = {
    'Authorization': f'Bearer {os.environ[\"APIDECK_API_KEY\"]}',
    'x-apideck-app-id': os.environ['APIDECK_APP_ID'],
    'x-apideck-consumer-id': os.environ['APIDECK_CONSUMER_ID'],
    'x-apideck-service-id': 'github',
    'Accept': 'application/json',
}

cursor = None
v2_with_runid = []
v1_with_runid = []
while True:
    url = f'https://unify.apideck.com/issue-tracking/collections/{os.environ[\"APIDECK_ISSUE_TRACKING_COLLECTION_ID\"]}/tickets?limit=100'
    if cursor:
        url += f'&cursor={cursor}'
    req = urllib.request.Request(url, method='GET', headers=headers)
    with urllib.request.urlopen(req) as resp:
        payload = json.loads(resp.read().decode('utf-8'))
    for t in payload.get('data', []):
        subj = t.get('subject') or ''
        if run_id in subj:
            if '[UPDATE-V2]' in subj:
                v2_with_runid.append(t)
            elif '[UPDATE-V1]' in subj:
                v1_with_runid.append(t)
    cursor = payload.get('meta', {}).get('cursors', {}).get('next')
    if not cursor:
        break

print(f'Run-id: {run_id}')
print(f'Tickets with [UPDATE-V2] + run-id: {len(v2_with_runid)}')
for t in v2_with_runid:
    print(f'  id={t[\"id\"]} subject={t[\"subject\"]!r} description={t[\"description\"]!r}')
print(f'Tickets with [UPDATE-V1] + run-id: {len(v1_with_runid)}')
for t in v1_with_runid:
