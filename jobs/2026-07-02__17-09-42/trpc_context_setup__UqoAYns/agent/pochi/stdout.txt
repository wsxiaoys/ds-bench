**`src/app/api/trpc/[trpc]/route.ts`** — App Router handler that uses `fetchRequestHandler` from `@trpc/server/adapters/fetch`. The `createContext` function receives the standard Web `Request` (passed in via closure from the handler wrapper) and reads the `x-user-id` header with `req.headers.get(...)`, falling back to `'anonymous'`. Both `GET` and `POST` are exported so tRPC's batch queries and mutations work.

```ts
import { fetchRequestHandler } from '@trpc/server/adapters/fetch';
import { appRouter } from '@/trpc/router';
import type { Context } from '@/trpc/init';

const createContext = async (req: Request): Promise<Context> => {
  const userId = req.headers.get('x-user-id') ?? 'anonymous';
  return { userId };
};

const handler = (req: Request) =>
  fetchRequestHandler({
    endpoint: '/api/trpc',
    req,
    router: appRouter,
    createContext: () => createContext(req),
    onError:
      process.env.NODE_ENV === 'development'
        ? ({ path, error }) => {
            console.error(
              `❌ tRPC failed on ${path ?? '<no-path>'}: ${error.message}`,
            );
          }
        : undefined,
  });

export { handler as GET, handler as POST };
```
