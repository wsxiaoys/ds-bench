```go
package main

import (
    "log"
    "math"
    "strings"

    "github.com/pocketbase/pocketbase"
    "github.com/pocketbase/pocketbase/core"

    _ "pbapp/migrations"
)

const wordsPerMinuteBatch = 200

func computeArticleMetrics(content string) (int, int) {
    words := len(strings.Fields(content))
    if words == 0 {
        return 0, 0
    }
    return words, int(math.Ceil(float64(words) / wordsPerMinuteBatch))
}

func writeArticleMetrics(record *core.Record) {
    words, minutes := computeArticleMetrics(record.GetString("content"))
    record.Set("word_count", words)
    record.Set("reading_time_minutes", minutes)
}

func main() {
    app := pocketbase.New()

    app.OnRecordCreateRequest("articles").BindFunc(func(e *core.RecordRequestEvent) error {
        writeArticleMetrics(e.Record)
        return e.Next()
    })

    app.OnRecordUpdateRequest("articles").BindFunc(func(e *core.RecordRequestEvent) error {
        writeArticleMetrics(e.Record)
        return e.Next()
    })

    if err := app.Start(); err != nil {
        log.Fatal(err)
    }
}
```

### How it meets the requirements
- **Both create and update** REST hooks registered with `OnRecordCreateRequest("articles")` and `OnRecordUpdateRequest("articles")`.
- **`word_count`** computed via `len(strings.Fields(content))` — empty or whitespace-only content yields `0`.
- **`reading_time_minutes`** computed via `int(math.Ceil(float64(words) / 200))`; with the `words == 0` early return, 0 words → 0 minutes.
- **Overwrites client values**: `record.Set("word_count", ...)` and `record.Set("reading_time_minutes", ...)` are called *after* the request payload has been merged into `e.Record`, so any values the client tried to set for those fields are replaced.
- **Chain continues**: each handler returns `e.Next()` so the record is persisted (verified — no records get stuck, server log is clean).
