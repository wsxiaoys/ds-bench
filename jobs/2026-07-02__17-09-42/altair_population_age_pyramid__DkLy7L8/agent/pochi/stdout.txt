### How the chart is built

The script uses `data.population.url` from `vega_datasets` (the canonical URL string, not a pre-loaded DataFrame), so type shorthands (`:O`, `:Q`, `:N`) are required:

```python
year_param = alt.param(
    name="year",
    value=1980,
    bind=alt.binding_select(options=YEAR_OPTIONS, name="Census year: "),
)

chart = (
    alt.Chart(population_url)                                 # URL, not a frame
    .transform_calculate(signed="datum.sex === 1 ? -datum.people : datum.people")
    .transform_filter("datum.year === year")                  # bound to the param
    .mark_bar()                                               # single bar mark
    .encode(
        y=alt.Y("age:O").title("Age").sort("descending"),     # oldest at top
        x=alt.X("signed:Q").title("Population").axis(
            format="s", labelExpr="abs(datum.value)"          # labels as absolute values
        ),
        color=alt.Color("sex:N").title("Sex").scale(
            domain=[1, 2], range=["steelblue", "salmon"]      # custom sex -> color
        ).legend(labelExpr="datum.value === 1 ? 'Male' : datum.value === 2 ? 'Female' : ''"),
    )
    .add_params(year_param)
    .properties(title="US Population Age Pyramid", width=420, height=320)
)
```
