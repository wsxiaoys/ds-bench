**`server.ts`** — Uses `initTRPC.create()` and the standalone adapter:
```ts
import { initTRPC } from '@trpc/server';
import { createHTTPServer } from '@trpc/server/adapters/standalone';

const t = initTRPC.create();

const appRouter = t.router({
  chatStream: t.procedure.query(async function* () {
    yield 'hello';
    yield 'world';
    yield '!';
  }),
});

export type AppRouter = typeof appRouter;

const server = createHTTPServer({ router: appRouter });
server.listen(3000);
```

**`client.ts`** — Uses `createTRPCClient` with `httpBatchStreamLink` and iterates the returned async iterable:
```ts
import { createTRPCClient, httpBatchStreamLink } from '@trpc/client';
import type { AppRouter } from './server.js';

const trpc = createTRPCClient<AppRouter>({
  links: [httpBatchStreamLink({ url: 'http://localhost:3000' })],
});

async function main() {
  const stream = await trpc.chatStream.query();
  for await (const chunk of stream) {
    console.log(chunk);
  }
}

main().catch((err) => { console.error(err); process.exit(1); });
```
