# Edge cases and more thorough testing

# Polygon with pentagon (CCW)
POLY pentagon 5 0 7.5 1.5 6.5 5 3.5 5 2.5 1.5

# Polygon with negative coords
POLY neg  -5 -5 5 -5 5 5 -5 5

# Polygon with very small values
POLY tiny 0 0 0.001 0 0.001 0.001 0 0.001

# Larger polygon (hexagon)
POLY hex 1 0 0.5 0.866 -0.5 0.866 -1 0 -0.5 -0.866 0.5 -0.866

# Pentagon area
AREA pentagon
# Neg square area = 100 (5*5*4 = 100, square area positive)
AREA neg
# Tiny area = 1e-6 = 0.000
AREA tiny
# Hexagon area = (3*sqrt(3)/2) ~ 2.598
AREA hex

# Pentagon with itself = overlap true
OVERLAP pentagon pentagon
# Pentagon with neg = depends on geometry
OVERLAP pentagon neg

# Point containment on hex
CONTAINS hex 0 0
CONTAINS hex 1 0
CONTAINS hex 0 1

# Point containment with int-like coordinates on pentagon
CONTAINS pentagon 5 0
CONTAINS pentagon 4 2

# Many concurrent segments
SEGMENTS 0 0 10 0 5 -1 5 1
SEGMENTS 0 0 10 0 5 -1 5 0
SEGMENTS 0 0 10 0 5 0 5 1
