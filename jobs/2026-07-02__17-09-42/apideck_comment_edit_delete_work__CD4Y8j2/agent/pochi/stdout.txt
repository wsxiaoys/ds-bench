headers = {

    "x-apideck-service-id": "github",
    "Accept": "application/json",
}

def get(path):
    req = urllib.request.Request(f"https://unify.apideck.com{path}", headers=headers)
    with urllib.request.urlopen(req, timeout=60) as r:
        return json.loads(r.read().decode())

cursor = None

tickets_total = 0
pages = 0
while True:

    resp = get(f"/issue-tracking/collections/{COLL}/tickets{qs}")

    tickets_total += len(data)
    pages += 1
    for t in data:
        if RID in (t.get("subject") or ""):
            matches.append(t)
    nxt = (resp.get("meta") or {}).get("cursors", {}).get("next")
    if not nxt:
        break
    cursor = nxt

print(f"pages={pages} total_tickets={tickets_total} matches_with_run_id={len(matches)}")

    print(t["id"], repr(t.get("subject")))
