    'Authorization': f'Bearer {API_KEY}',
    'x-apideck-app-id': APP_ID,
    'x-apideck-consumer-id': CONSUMER_ID,
    'x-apideck-service-id': 'onedrive',
}
PREFIX = 'AGG-' + open('/logs/artifacts/run-id').read().strip() + '-'
print('Looking for prefix:', PREFIX)
cursor = None
all_names = set()
while True:
    params = {'limit': 50}
    if cursor:
        params['cursor'] = cursor
    r = requests.get('https://unify.apideck.com/file-storage/files', headers=headers, params=params, timeout=60)
    body = r.json()
    for item in body.get('data', []):
        all_names.add(item.get('name'))
    nxt = (body.get('meta') or {}).get('cursors', {}).get('next')
    if not nxt:
        break
    cursor = nxt
matches = [n for n in all_names if n.startswith(PREFIX)]
print('Total files seen:', len(all_names))
print('Matching prefix:', matches)
