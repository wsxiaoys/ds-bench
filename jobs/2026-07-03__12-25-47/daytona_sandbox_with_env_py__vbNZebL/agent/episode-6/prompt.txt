import os
from daytona import Daytona, CreateSandboxFromSnapshotParams


def main():
    # Read run-id
    with open('/logs/artifacts/run-id', 'r') as f:
        run_id = f.read().strip()

    sandbox_name = f"envvar-py-{run_id}"
    my_var_value = f"hello-{run_id}"

    daytona = Daytona()
    sandbox = None
    log_lines = []

    try:
        # Create sandbox with custom env vars
        params = CreateSandboxFromSnapshotParams(
            name=sandbox_name,
            env_vars={
                "MY_VAR": my_var_value,
                "APP_MODE": "production",
            },
        )
        sandbox = daytona.create(params)

        # Read MY_VAR from sandbox
        resp_my_var = sandbox.process.exec("echo \"$MY_VAR\"")
        my_var_captured = resp_my_var.result.strip()
        log_lines.append(f"MY_VAR: {my_var_captured}")

        # Read APP_MODE from sandbox
        resp_app_mode = sandbox.process.exec("echo \"$APP_MODE\"")
        app_mode_captured = resp_app_mode.result.strip()
        log_lines.append(f"APP_MODE: {app_mode_captured}")

        # Write to log file
        with open('/home/user/myproject/output.log', 'w') as f:
            f.write("\n".join(log_lines) + "\n")

        print("Log file written:")
        print("\n".join(log_lines))
    finally:
        if sandbox is not None:
            try:
                sandbox.delete()
            except Exception as e:
                print(f"Failed to delete sandbox: {e}")


if __name__ == "__main__":
    main()
