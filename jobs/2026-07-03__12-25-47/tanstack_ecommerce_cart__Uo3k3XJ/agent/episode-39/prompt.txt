  productId: number
  quantity: number
}

type CartDetail = {
  product: Product

  subtotal: number

function parseCart(cart: string | undefined): CartItem[] {
  if (!cart) return []
  return cart
    .split(',')
    .map((part) => part.trim())
    .filter(Boolean)
    .map((part): CartItem | null => {
      const [idStr, qtyStr] = part.split(':')
      const productId = Number(idStr)
      const quantity = Number(qtyStr) || 0
      if (!Number.isFinite(productId) || quantity <= 0) return null
      return { productId, quantity }
    })
    .filter((x): x is CartItem => x !== null)
}

function serializeCart(items: CartItem[]): string {
  return items.map((i) => `${i.productId}:${i.quantity}`).join(',')
}
