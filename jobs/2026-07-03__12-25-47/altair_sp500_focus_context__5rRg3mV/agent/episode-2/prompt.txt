import altair as alt
from vega_datasets import data

# Data source: use the URL directly from vega_datasets (no pandas processing)
source = data.sp500.url

# Brush selection restricted to x-axis, used by the lower (context) chart
brush = alt.selection_interval(encodings=['x'])

# Lower (context) chart: small overview with brush
context = (
    alt.Chart(source)
    .mark_area()
    .encode(
        x=alt.X('date:T', title='Date'),
        y=alt.Y('price:Q', title='Price'),
    )
    .properties(
        height=60,
        title='S&P 500 Overview (drag to brush)',
    )
    .add_params(brush)
)

# Upper (detail) chart: large area whose x-domain follows the brush
detail = (
    alt.Chart(source)
    .mark_area()
    .encode(
        x=alt.X(
            'date:T',
            title='Date',
            scale=alt.Scale(domain=brush),
        ),
        y=alt.Y('price:Q', title='Price'),
    )
    .properties(
        height=300,
        title='S&P 500 Detail',
    )
)

# Horizontal rule marking the running maximum price inside the brushed window.
# Computed inside the spec: filter by brush, then aggregate max of price.
max_rule = (
    alt.Chart(source)
    .transform_filter(brush)
    .transform_aggregate(max_price='max(price)')
    .mark_rule(color='firebrick', size=2)
    .encode(y='max_price:Q')
)

# Layer the rule onto the detail chart, then concatenate detail over context
detail_with_rule = alt.layer(detail, max_rule)

chart = detail_with_rule & context

if __name__ == '__main__':
    chart.save('/home/user/altair_project/chart.html')
    print('Saved /home/user/altair_project/chart.html')
