extends Node3D

const LEVEL_PATH := "res://data/level.json"

var grid_map: GridMap

func _ready() -> void:
    grid_map = get_node_or_null("GridMap") as GridMap

func _ensure_grid_map() -> GridMap:
    if grid_map == null:
        grid_map = get_node_or_null("GridMap") as GridMap
    return grid_map

func _make_box_mesh(color: Color) -> BoxMesh:
    var mesh := BoxMesh.new()
    mesh.size = Vector3(1.0, 1.0, 1.0)
    var mat := StandardMaterial3D.new()
    mat.albedo_color = color
    mesh.surface_set_material(0, mat)
    mesh.material = mat
    return mesh

func _build_mesh_library() -> MeshLibrary:
    var lib := MeshLibrary.new()
    for id in range(3):
        lib.create_item(id)
    var color0 := Color(0.9, 0.1, 0.1)
    var color1 := Color(0.1, 0.9, 0.1)
    var color2 := Color(0.1, 0.3, 0.95)
    lib.set_item_mesh(0, _make_box_mesh(color0))
    lib.set_item_mesh(1, _make_box_mesh(color1))
    lib.set_item_mesh(2, _make_box_mesh(color2))
    var shape := BoxShape3D.new()
    shape.size = Vector3(1.0, 1.0, 1.0)
    for id in range(3):
        lib.set_item_shapes(id, [shape])
    return lib

func _load_level_cells() -> Array:
    var f := FileAccess.open(LEVEL_PATH, FileAccess.READ)
    if f == null:
        push_error("Failed to open level file: %s (err %d)" % [LEVEL_PATH, FileAccess.get_open_error()])
        return []
    var text := f.get_as_text()
    f.close()
    var parsed: Variant = JSON.parse_string(text)
    if typeof(parsed) != TYPE_DICTIONARY:
        push_error("level.json is not a JSON object")
        return []
    var cells_v: Variant = parsed.get("cells", [])
    if typeof(cells_v) != TYPE_ARRAY:
        return []
    return cells_v

func build() -> void:
    var gm := _ensure_grid_map()
    if gm == null:
        push_error("GridMap child node not found")
        return
    var lib := _build_mesh_library()
    gm.mesh_library = lib
    gm.clear()
    var cells: Array = _load_level_cells()
    for entry_v in cells:
        if typeof(entry_v) != TYPE_DICTIONARY:
            continue
        var entry: Dictionary = entry_v
        var id_i: int = int(entry.get("id", -1))
        var x: int = int(entry.get("x", 0))
        var y: int = int(entry.get("y", 0))
        var z: int = int(entry.get("z", 0))
        if id_i < 0 or id_i > 2:
            continue
        gm.set_cell_item(Vector3i(x, y, z), id_i)

func place(item_id: int, gx: int, gy: int, gz: int) -> void:
    var gm := _ensure_grid_map()
    if gm == null:
        return
    gm.set_cell_item(Vector3i(gx, gy, gz), item_id)

func remove(gx: int, gy: int, gz: int) -> void:
    var gm := _ensure_grid_map()
    if gm == null:
        return
    gm.set_cell_item(Vector3i(gx, gy, gz), GridMap.INVALID_CELL_ITEM)

func get_item(gx: int, gy: int, gz: int) -> int:
    var gm := _ensure_grid_map()
    if gm == null:
        return -1
    return int(gm.get_cell_item(Vector3i(gx, gy, gz)))
