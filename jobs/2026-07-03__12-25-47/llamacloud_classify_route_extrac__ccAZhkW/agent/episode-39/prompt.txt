      confidence: c.cat.confidence,
      file_id: c.fileId,
      data,

      `Routed: ${c.basename} | category: ${c.cat.type} | confidence: ${c.cat.confidence} | fields: ${nFields}`,
    );
  }

  fs.writeFileSync(
    path.join(outputsDir, 'results.json'),

  );
