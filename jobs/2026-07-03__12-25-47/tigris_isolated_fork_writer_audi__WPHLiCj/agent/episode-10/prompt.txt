#!/bin/bash
set -e

# Read run id
RUN_ID=$(cat /logs/artifacts/run-id)
BUCKET_NAME="harbor-isolation-${RUN_ID}"

# Normalize: lowercase and replace invalid chars with hyphens
BUCKET_NAME=$(echo "$BUCKET_NAME" | tr '[:upper:]' '[:lower:]' | sed 's/[^a-z0-9.-]/-/g')

cd /home/user/isolation

# Try to create the bucket (idempotent)
node --experimental-vm-modules -e "
import('@tigrisdata/storage').then(async ({ createBucket, putObject, getBucketMetadata, setBucketSnapshot }) => {
  const { data, error } = await createBucket(process.env.BUCKET_NAME, { enableSnapshot: true });
  if (error && !String(error.message).includes('already exists')) {
    console.error('Bucket creation failed:', error.message);
    process.exit(1);
  }
  console.log('Bucket ready');
  await putObject(process.env.BUCKET_NAME, 'seed1.txt', Buffer.from('seed1 content'));
  await putObject(process.env.BUCKET_NAME, 'seed2.txt', Buffer.from('seed2 content'));
  console.log('Seeds uploaded');
});
" 2>&1 || true
