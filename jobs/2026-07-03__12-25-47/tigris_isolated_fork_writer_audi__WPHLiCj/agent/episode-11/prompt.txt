#!/bin/bash
# Setup script for isolation task. Idempotent.
set -e

RUN_ID=$(cat /logs/artifacts/run-id)
BUCKET_NAME="harbor-isolation-${RUN_ID}"
# Normalize: lowercase and replace invalid chars with hyphens
BUCKET_NAME=$(echo "$BUCKET_NAME" | tr '[:upper:]' '[:lower:]' | sed 's/[^a-z0-9.-]/-/g')

export BUCKET_NAME

cd /home/user/isolation

node --input-type=module -e "
import { createBucket, putObject } from '@tigrisdata/storage';
const bucketName = process.env.BUCKET_NAME;
console.log('Setting up bucket:', bucketName);

const { data, error } = await createBucket(bucketName, { enableSnapshot: true });
if (error) {
  if (String(error.message).toLowerCase().includes('already exists')) {
    console.log('Bucket already exists, continuing');
  } else {
    console.error('Bucket creation error:', error.message);
    process.exit(1);
  }
} else {
  console.log('Bucket created');
}

const r1 = await putObject(bucketName, 'seed1.txt', Buffer.from('seed1 content\n'));
if (r1.error) { console.error('putObject seed1 error:', r1.error.message); process.exit(1); }
console.log('seed1.txt uploaded');

const r2 = await putObject(bucketName, 'seed2.txt', Buffer.from('seed2 content\n'));
if (r2.error) { console.error('putObject seed2 error:', r2.error.message); process.exit(1); }
console.log('seed2.txt uploaded');

console.log('Setup complete for bucket:', bucketName);
"
