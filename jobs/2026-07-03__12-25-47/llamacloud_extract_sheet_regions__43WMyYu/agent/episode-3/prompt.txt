import os
from pathlib import Path
from llama_cloud import LlamaCloud
import urllib.request

DATA_DIR = Path("/home/user/project")
XLSX_PATH = DATA_DIR / "data" / "sales.xlsx"
OUTPUT_DIR = DATA_DIR / "output"
LOG_PATH = OUTPUT_DIR / "sheets.log"

OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

client = LlamaCloud()

with open(XLSX_PATH, "rb") as f:
    upload = client.files.create(file=f, purpose="parse")
file_id = upload.id
print(f"Uploaded file: {file_id}")

job = client.beta.sheets.parse(
    file_id=file_id,
    config={"generate_additional_metadata": True},
)
print(f"Job id: {job.id} status: {job.status}")
print(f"Regions: {job.regions}")
print(f"Worksheet metadata: {getattr(job, 'worksheet_metadata', None)}")

# Build a map of region_id -> (sheet_name, location) from worksheet metadata
region_info = {}
wm = getattr(job, "worksheet_metadata", None)
if wm:
    for ws in wm:
        for r in (getattr(ws, "regions", None) or []):
            region_info[r.region_id] = (
                getattr(ws, "title", "") or getattr(ws, "name", ""),
                getattr(r, "location", "") or getattr(r, "range", ""),
            )

log_lines = []
log_lines.append(f"Job ID: {job.id}")
log_lines.append(f"Job Status: {job.status}")
log_lines.append(f"Region Count: {len(job.regions)}")

for region in job.regions:
    sheet_name, location = region_info.get(region.region_id, ("", ""))
    # fallback: try attributes on region itself
    if not location:
        location = getattr(region, "location", "") or getattr(region, "range", "")
    if not sheet_name:
        sheet_name = getattr(region, "sheet_name", "") or getattr(region, "worksheet_title", "")

    parquet_path = OUTPUT_DIR / f"region_{region.region_id}.parquet"

    table = client.beta.sheets.get_result_table(
        region_type=region.region_type,
        spreadsheet_job_id=job.id,
        region_id=region.region_id,
    )
    url = table.url
    with urllib.request.urlopen(url) as resp, open(parquet_path, "wb") as out:
        out.write(resp.read())

    log_lines.append(
        f"Region: {region.region_id} sheet={sheet_name} location={location}"
    )
    log_lines.append(f"Parquet: {parquet_path}")

LOG_PATH.write_text("\n".join(log_lines) + "\n")
print(f"Wrote log to {LOG_PATH}")
print("\n".join(log_lines))
