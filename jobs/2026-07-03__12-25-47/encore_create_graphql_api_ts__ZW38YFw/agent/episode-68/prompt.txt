export const graphqlAPI = api.raw(
    {expose: true, path: "/graphql", method: "*"},

        server.assertStarted("/graphql");

        const headers = new HeaderMap();

                headers.set(key, Array.isArray(value) ? value.join(", ") : value);
            }
        }

        })
