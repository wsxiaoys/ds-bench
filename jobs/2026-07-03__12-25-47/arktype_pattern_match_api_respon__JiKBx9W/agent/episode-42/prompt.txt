import { match, scope } from 'arktype';

// Define a scope with the discriminated union types
const $ = scope({
  success: { status: '"success"', data: 'object' },
  error: { status: '"error"', code: 'number', reason: 'string' },
  pending: { status: '"pending"' },
});

const handle = $.match({
  success: (data) => `OK: ${JSON.stringify((data as any).data)}`,
  error: (e) => `ERR ${(e as any).code} ${(e as any).reason}`,
  pending: () => `PENDING`,
  default: "assert"
});

// Read input from stdin
const chunks: Buffer[] = [];
process.stdin.on('data', (chunk) => chunks.push(chunk));
process.stdin.on('end', () => {
  try {
    const input = JSON.parse(Buffer.concat(chunks).toString('utf8'));
    const result = handle(input);
    process.stdout.write(result + '\n');
  } catch (err) {
    process.exit(1);
  }
});
