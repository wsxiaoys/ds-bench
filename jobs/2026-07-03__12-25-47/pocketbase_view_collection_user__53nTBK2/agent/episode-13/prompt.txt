func (c *Collection) initDefaultFields() {
        switch c.Type {
        case CollectionTypeBase:
                c.initIdField()
        case CollectionTypeAuth:
                c.initIdField()
                c.initPasswordField()
                c.initTokenKeyField()
                c.initEmailField()
                c.initEmailVisibilityField()
                c.initVerifiedField()
        case CollectionTypeView:
                // view fields are REDACTEDgenerated
        }
}

func (c *Collection) initIdField() {
        field, _ := c.Fields.GetByName(FieldNameId).(*TextField)
        if field == nil {
                // create default field
                field = &TextField{
                        Name:                FieldNameId,
                        System:              true,
                        PrimaryKey:          true,
                        Required:            true,
                        Min:                 15,
                        Max:                 15,
                        Pattern:             defaultLowercaseRecordIdPattern,
                        AutogeneratePattern: `[a-z0-9]{15}`,
                }

                // prepend it
                c.Fields = NewFieldsList(append([]Field{field}, c.Fields...)...)
        } else {
                // enforce system defaults
                field.System = true
                field.Required = true
                field.PrimaryKey = true
                field.Hidden = false
                if field.Pattern == "" {
                        field.Pattern = defaultLowercaseRecordIdPattern
                }
        }
}
