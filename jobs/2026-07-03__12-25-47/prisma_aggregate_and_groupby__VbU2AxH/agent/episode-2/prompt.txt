const fs = require('fs');

const prisma = new PrismaClient();

async function main() {
  const totals = await prisma.order.aggregate({
    _count: true,
    _sum: { amount: true },
    _avg: { amount: true },
  });

  const byStatus = await prisma.order.groupBy({
    by: ['status'],
    _count: true,
    _sum: { amount: true },
  });

  const result = {
    totals: {
