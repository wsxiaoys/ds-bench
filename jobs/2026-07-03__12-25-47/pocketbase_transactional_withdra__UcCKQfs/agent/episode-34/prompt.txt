package main

import (
"errors"
"log"

"github.com/pocketbase/pocketbase"
"github.com/pocketbase/pocketbase/core"
"github.com/pocketbase/pocketbase/plugins/migratecmd"

_ "myproject/migrations"
)

func main() {
app := pocketbase.New()

migratecmd.MustRegister(app, app.RootCmd, migratecmd.Config{
Automigrate: false,
})

app.OnRecordCreateRequest("withdrawals").BindFunc(func(e *core.RecordRequestEvent) error {
walletId := e.Record.GetString("wallet")
amount := e.Record.GetFloat("amount")

if walletId == "" || amount <= 0 {
return e.BadRequestError("Invalid withdrawal payload", nil)
}

// We need to look up the wallet to verify it exists and has enough balance.
// We do this in a transaction so the debit and the withdrawal persist atomically.
return app.RunInTransaction(func(txApp core.App) error {
wallet, err := txApp.FindRecordById("wallets", walletId)
if err != nil {
return e.BadRequestError("Wallet not found", nil)
}

balance := wallet.GetFloat("balance")
if balance < amount {
return e.BadRequestError("Insufficient funds", nil)
}

wallet.Set("balance", balance-amount)
if err := txApp.Save(wallet); err != nil {
return err
}

// Repoint e.App at the transactional app so the default save chain
// (triggered by e.Next()) runs inside the same transaction.
e.App = txApp

return e.Next()
})
})

if err := app.Start(); err != nil {
log.Fatal(err)
}
}
