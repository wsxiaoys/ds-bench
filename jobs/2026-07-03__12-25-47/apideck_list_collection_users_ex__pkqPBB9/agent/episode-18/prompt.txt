import json
import os
from apideck_unify import Apideck
from apideck_unify.types import UNSET


def to_value(field):
    """Convert UNSET or None to None, otherwise return the value as a string."""
    if field is None or field is UNSET:
        return None
    return str(field)


def main():
    api_key = os.environ.get("APIDECK_API_KEY")
    app_id = os.environ.get("APIDECK_APP_ID")
    consumer_id = os.environ.get("APIDECK_CONSUMER_ID")
    collection_id = os.environ.get("APIDECK_ISSUE_TRACKING_COLLECTION_ID")

    apideck = Apideck(
        api_key=api_key,
        app_id=app_id,
        consumer_id=consumer_id,
    )

    all_users = []
    page = apideck.issue_tracking.collection_users.list(
        collection_id=collection_id,
        service_id="github",
        limit=200,
    )

    while page is not None:
        response_data = page.get_collection_users_response
        if response_data is not None and response_data.data is not None:
            for user in response_data.data:
                all_users.append(
                    {
                        "id": to_value(user.id),
                        "name": to_value(user.name),
                        "first_name": to_value(user.first_name),
                        "last_name": to_value(user.last_name),
                        "email": to_value(user.email),
                    }
                )
        page = page.next()

    output = {
        "collection_id": collection_id,
        "service_id": "github",
        "users": all_users,
    }

    with open("/home/user/myproject/users.json", "w") as f:
        json.dump(output, f, indent=2)

    with open("/home/user/myproject/output.log", "w") as f:
        f.write(f"User count: {len(all_users)}\n")

    print(f"Collected {len(all_users)} users")


if __name__ == "__main__":
    main()
