   from concurrent.futures import ThreadPoolExecutor
   import boto3
   from botocore.client import Config

   run_id = open("/logs/artifacts/run-id").read().strip()
   bucket = f"harbor-bulk-{run_id}"
   import re
   bucket = re.sub(r"[^a-z0-9.-]", "-", bucket.lower())

   )
