            Returns:

        @classmethod
        def get_models(cls, include_empty: bool = False) -> set[SQLModelOrSqlAlchemy]:
            """Get registered models.

            Args:
                include_empty: If True, include models with empty metadata.

            Returns:
                The registered models.
            """
