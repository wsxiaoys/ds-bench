>   });
>
>   const handleSubmit = (e) => {
>     e.preventDefault();
>     const trimmed = text.trim();

>     mutation.mutate(trimmed);

>   };
>
>   return (
>     <div>
>       <h1>Todo List</h1>
>       <form onSubmit={handleSubmit}>
>         <input
>           id="todo-input"

>           value={text}
>           onChange={(e) => setText(e.target.value)}
