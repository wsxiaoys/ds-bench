>
> Reads events from `input.jsonl`, groups by `user_id`, deduplicates per
> `(user_id, event_id)` pair using a 10-second time window (inclusive at
> exactly 10s), and writes the surviving events to `output.jsonl` in the
> original relative order.

> def _parse_ts(s: Any) -> datetime:
>     """Parse an ISO 8601 timestamp string into a tz-aware `datetime`."""
>     if isinstance(s, datetime):
>         return s
>     if s.endswith("Z"):
>         s = s[:-1] + "+00:00"
>     return datetime.fromisoformat(s)

> class DedupLogic(
>     StatefulLogic[Dict[str, Any], Dict[str, Any], Dict[str, datetime]]
> ):
>     """Per-user_id event-id deduplication with a sliding 10s window.
>
>     State is a mapping `event_id -> last_emitted_datetime` for the
>     current `user_id`. On every event:
>
>       1. Any `event_id` whose stored timestamp is older than 10 seconds
>          from the current event's timestamp is dropped from the state
>          so memory cannot grow without bound.
>       2. If the same `event_id` was emitted within the last 10 seconds
>          (inclusive of exactly 10s, i.e. `ts <= last + 10s`), the new
>          event is dropped.
>       3. Otherwise the event is emitted and the 10-second window is
>          restarted for that `event_id`.
>     """
>
>     def __init__(self, resume_state):
>         # Map of event_id -> last emitted timestamp (datetime).
>         self._state: Dict[str, datetime] = (
>             dict(resume_state) if resume_state else {}
>         )
>
>     def on_item(
>         self, value: Dict[str, Any]
>     ) -> Tuple[Iterable[Dict[str, Any]], bool]:
>         event_id = value["event_id"]
>         ts = _parse_ts(value["timestamp"])
>
>         # 1) Drop entries outside the sliding window (strictly older than
>         #    10s from the current event's timestamp).
>         stale = [
>             eid for eid, last_ts in self._state.items()
>             if ts - last_ts > WINDOW
>         ]
>         for eid in stale:
>             del self._state[eid]
>
>         # 2) Deduplication check against the freshly-cleared state.
>         prev_ts = self._state.get(event_id)
>         if prev_ts is not None and ts <= prev_ts + WINDOW:
>             # Duplicate within the window (or exactly at the boundary).
>             return (iter(()), StatefulLogic.RETAIN)
>
>         # 3) Emit and (re)start the 10-second window for this event_id.
>         self._state[event_id] = ts
>         return (iter((value,)), StatefulLogic.RETAIN)
>
>     def snapshot(self) -> Dict[str, datetime]:
>         # Must return an effectively immutable copy.
>         return dict(self._state)
>
>
> def _build_logic(resume_state):
>     return DedupLogic(resume_state)
