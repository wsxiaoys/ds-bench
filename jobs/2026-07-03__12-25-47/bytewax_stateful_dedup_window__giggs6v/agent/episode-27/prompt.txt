import json
from datetime import datetime, timedelta
from pathlib import Path
from typing import Any, Dict, Iterable, Tuple

from bytewax import operators as op
from bytewax.connectors.files import FileSource
from bytewax.dataflow import Dataflow
from bytewax.operators import StatefulLogic
from bytewax.outputs import StatelessSinkPartition, DynamicSink
from bytewax.testing import run_main
from typing_extensions import override

WINDOW = timedelta(seconds=10)


def parse_ts(s) -> datetime:
    """Parse an ISO 8601 timestamp string into a tz-aware datetime."""
    if isinstance(s, datetime):
        return s
    if s.endswith("Z"):
        s = s[:-1] + "+00:00"
    return datetime.fromisoformat(s)


def fmt_ts(ts: datetime) -> str:
    """Render a datetime back to an ISO 8601 string with trailing `Z`."""
    if ts.tzinfo is not None:
        ts = ts.astimezone(tz=ts.tzinfo).replace(microsecond=0)
        return ts.isoformat().replace("+00:00", "Z")
    return ts.isoformat()


class DedupLogic(StatefulLogic[Dict[str, Any], Dict[str, Any], Dict[str, datetime]]):
    """Per-user_id event-id deduplication over a sliding 10s window."""

    def __init__(self, resume_state):
        # `state` maps `event_id` -> last emitted timestamp (datetime).
        self._state: Dict[str, datetime] = dict(resume_state) if resume_state else {}

    def on_item(
        self, value: Dict[str, Any]
    ) -> Tuple[Iterable[Dict[str, Any]], bool]:
        event_id = value["event_id"]
        ts = parse_ts(value["timestamp"])

        # Clean up state: any event_id older than the window relative to
        # the current event's timestamp is removed.
        stale = [
            eid for eid, last_ts in self._state.items()
            if ts - last_ts > WINDOW
        ]
        for eid in stale:
            del self._state[eid]

        # Deduplication check: if the same event_id was seen within the
        # last 10 seconds (inclusive of exactly 10s), drop.
        prev_ts = self._state.get(event_id)
        if prev_ts is not None and ts <= prev_ts + WINDOW:
            return (iter(()), StatefulLogic.RETAIN)

        # Otherwise emit and (re)start the 10-second window for this
        # event_id.
        self._state[event_id] = ts
        return (iter((value,)), StatefulLogic.RETAIN)

    def snapshot(self) -> Dict[str, datetime]:
        # Must return an effectively immutable copy.
        return dict(self._state)


def build_logic(resume_state):
    return DedupLogic(resume_state)


class _JsonlSinkPartition(StatelessSinkPartition[str]):
    def __init__(self, path: Path):
        self._path = path
        self._items: list = []

    @override
    def write_batch(self, items) -> None:
        # Buffer everything; we'll sort then write on close.
        self._items.extend(items)

    @override
    def close(self) -> None:
        # Preserve input order using the embedded sequence number, then
        # drop the sequence number field and dump to JSONL.
        def _key(item):
            return item["__seq"]

        self._items.sort(key=_key)
        with self._path.open("wt") as f:
            for item in self._items:
                item.pop("__seq", None)
                f.write(json.dumps(item))
                f.write("\n")


class JsonlSink(DynamicSink[str]):
    """Custom Sink that buffers items, sorts them by a sequence number,
    and writes them as JSON lines to a single file."""

    def __init__(self, path):
        self._path = path

    @override
    def build(self, step_id, worker_index, worker_count):
        return _JsonlSinkPartition(Path(self._path))


def build_flow(input_path, output_path) -> Dataflow:
    flow = Dataflow("stateful_dedup")

    # 1) Read raw lines.
    lines = op.input("input", flow, FileSource(str(input_path)))

    # 2) Parse JSON, attach a sequence number to preserve input order.
    counter = {"i": 0}

    def _parse_with_seq(line):
        event = json.loads(line)
        event["__seq"] = counter["i"]
        counter["i"] += 1
        return event

    parsed = op.map("parse_json", lines, _parse_with_seq)

    # 3) Key by user_id so state is maintained per user.
    keyed = op.key_on("key_user", parsed, lambda e: e["user_id"])

    # 4) Apply stateful deduplication.
    deduped = op.stateful("dedup", keyed, build_logic)

    # 5) Drop key and serialize. The deduped stream is keyed (user_id, event).
    def _to_json(item):
        user_id, event = item
        out = dict(event)
        ts = out["timestamp"]
        if isinstance(ts, datetime):
            out["timestamp"] = fmt_ts(ts)
        return out

    events = op.map("to_dict", deduped, _to_json)

    # 6) Write out (the custom sink handles sequencing + JSONL serialization).
    op.output("output", events, JsonlSink(output_path))

    return flow


if __name__ == "__main__":
    run_main(build_flow(Path("input.jsonl"), Path("output.jsonl")))
