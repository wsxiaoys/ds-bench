def build_flow(input_path, output_path) -> Dataflow:
    flow = Dataflow("stateful_dedup")

    # 1) Read raw lines from the input file.
    lines = op.input("input", flow, FileSource(str(input_path)))

    # 2) Parse each JSON line into a dict, attaching a sequence number
    #    used later to recover input ordering after the keyed/stateful
    #    step (which may interleave keys across workers).
    counter = {"i": 0}

    def _parse_with_seq(line):
        event = json.loads(line)
        event["__seq"] = counter["i"]
        counter["i"] += 1
        return event

    parsed = op.map("parse_json", lines, _parse_with_seq)

    # 3) Key by user_id so the per-key state is partitioned per user.
    keyed = op.key_on("key_user", parsed, lambda event: event["user_id"])

    # 4) Apply stateful deduplication keyed on user_id.
    deduped = op.stateful("dedup", keyed, _build_logic)

    # 5) Drop the key produced by `stateful` and forward the event dict.
    unkeyed = op.key_rm("unkey", deduped)

    # 6) Write out as JSONL preserving input order.
    op.output("output", unkeyed, OrderedJsonlSink(output_path))

    return flow


if __name__ == "__main__":
    run_main(build_flow(Path("input.jsonl"), Path("output.jsonl")))
