// Parameter type: object with url, timeoutMs, retries
export const fetchParams = type({
  url: 'string.url',
  timeoutMs: '0 < number.integer <= 10000',
  retries: '0 <= number.integer <= 5',
});

// Resolved-value type: object with status, body
export const fetchResponse = type({
  status: '100 <= number.integer <= 599',
  body: 'string',
});

// A Promise schema that validates the resolved value via the fetchResponse type
// and propagates the validation error to the returned promise.
export const fetchPromise = type('Promise').pipe((p: Promise<unknown>) => {
  return (p as Promise<unknown>).then((v) => {
    const validated = fetchResponse(v);
    if (validated instanceof Error) throw validated;
    return validated;
  });
});

// Validated async fetch wrapper.
// Parameter validation happens synchronously at the type.fn boundary.
export const fetchWithTimeout = type.fn(
  fetchParams,
  ':',
  fetchPromise,
)(async (params) => {
  // setTimeout is scheduled synchronously when this async body begins executing.
  // Parameter validation has already happened by this point.
  const delay = Math.min(params.timeoutMs, 50);
  await new Promise<void>((resolve) => setTimeout(resolve, delay));
  return { status: 200, body: 'ok' };
});

export default fetchWithTimeout;
