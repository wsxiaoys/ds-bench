func generate(target: TileMapLayer) -> void:
    _rooms.clear()
    if target == null:
        return
    target.clear()

    # 1. Noise pass

    noise.seed = seed

    noise.frequency = 0.1

    for y in range(height):

            var n: float = noise.get_noise_2d(x, y)
            # n is in [-1, 1]; wall_threshold default 0.0 -> n >= 0 is floor
            if n >= wall_threshold:
                target.set_cell(Vector2i(x, y), 0, Vector2i(0, 0))
            else:
                target.set_cell(Vector2i(x, y), 1, Vector2i(1, 0))

        target.set_cell(Vector2i(0, y), 1, Vector2i(1, 0))
        target.set_cell(Vector2i(width - 1, y), 1, Vector2i(1, 0))

    var rng := RandomNumberGenerator.new()

    var room_count: int = 4
    var max_attempts: int = 64
    var attempts: int = 0
    var min_size: int = 4
    var max_size: int = 8

    while _rooms.size() < room_count and attempts < max_attempts:
        attempts += 1
        var rw: int = rng.randi_range(min_size, max_size)
        var rh: int = rng.randi_range(min_size, max_size)
        var rx: int = rng.randi_range(1, width - 2 - rw)
        var ry: int = rng.randi_range(1, height - 2 - rh)
        var rect := Rect2i(rx, ry, rw, rh)
        var ok: bool = true

            if r.intersects(rect):

                break

        _rooms.append(rect)
        # Carve the room (write floor)
        for y2 in range(ry, ry + rh):

            y0 += step_y
        target.set_cell(Vector2i(c_center.x, c_center.y), 0, Vector2i(0, 0))
