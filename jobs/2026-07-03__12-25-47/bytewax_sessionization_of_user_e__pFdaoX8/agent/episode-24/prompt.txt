from typing_extensions import override


class _PrintSinkPartition(StatelessSinkPartition[Any]):
    @override
    def write_batch(self, items: List[Any]) -> None:
        for item in items:
            line = str(item)
            sys.stdout.write(line)
            sys.stdout.write("\n")
        sys.stdout.flush()
