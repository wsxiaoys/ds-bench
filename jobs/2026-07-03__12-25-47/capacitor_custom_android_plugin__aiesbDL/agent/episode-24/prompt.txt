package com.example.myapp;

import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;

@CapacitorPlugin(name = "DeviceSensor")
public class DeviceSensorPlugin extends Plugin {

    @PluginMethod
    public void getReading(PluginCall call) {
        String sensor = call.getString("sensor");
        if (sensor == null) {
            call.reject("Missing 'sensor' option");
            return;
        }

        JSObject ret = new JSObject();
        switch (sensor) {
            case "temperature":
                ret.put("sensor", "temperature");
                ret.put("value", 22.5);
                ret.put("unit", "C");
                call.resolve(ret);
                break;
            case "humidity":
                ret.put("sensor", "humidity");
                ret.put("value", 65.0);
                ret.put("unit", "%");
                call.resolve(ret);
                break;
            case "battery":
                ret.put("sensor", "battery");
                ret.put("value", 87.0);
                ret.put("unit", "%");
                call.resolve(ret);
                break;
            default:
                call.reject("Unsupported sensor: " + sensor);
                break;
        }
    }

    @PluginMethod
    public void isAvailable(PluginCall call) {
        String sensor = call.getString("sensor");
        JSObject ret = new JSObject();
        if (sensor == null) {
            ret.put("available", false);
            call.resolve(ret);
            return;
        }

        switch (sensor) {
            case "temperature":
            case "humidity":
            case "battery":
                ret.put("available", true);
                break;
            default:
                ret.put("available", false);
                break;
        }
        call.resolve(ret);
    }
}

import { registerPlugin } from '@capacitor/core';

export interface DeviceSensorReading {
  sensor: string;
  value: number;
  unit: string;
}

export interface DeviceSensorAvailability {
  available: boolean;
}

export interface DeviceSensorPlugin {
  getReading(options: { sensor: string }): Promise<DeviceSensorReading>;
  isAvailable(options: { sensor: string }): Promise<DeviceSensorAvailability>;
}

const DeviceSensor = registerPlugin<DeviceSensorPlugin>('DeviceSensor');

export default DeviceSensor;
