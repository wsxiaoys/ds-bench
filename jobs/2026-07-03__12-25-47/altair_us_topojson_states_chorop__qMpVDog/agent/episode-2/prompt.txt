import altair as alt
from vega_datasets import data

# Load TopoJSON URL for US state boundaries
states = alt.topo_feature(data.us_10m.url, 'states')

# URL for population_engineers_hurricanes dataset
pop_url = data.population_engineers_hurricanes.url

# Build the chart
chart = (
    alt.Chart(states)
    .mark_geoshape()
    .transform_lookup(
        lookup='id',
        from_=alt.LookupData(
            pop_url,
            key='id',
            fields=['engineers', 'state'],
        ),
    )
    .encode(
        color=alt.Color('engineers:Q', scale=alt.Scale(scheme='blues')),
        tooltip=['state:N', 'engineers:Q'],
    )
    .project(type='albersUsa')
    .properties(width=700, height=400)
)

chart.save('chart.html')
print('Chart saved to chart.html')
