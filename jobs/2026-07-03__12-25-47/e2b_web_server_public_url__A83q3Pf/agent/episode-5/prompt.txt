import json
import time
import os
from e2b import Sandbox

# Step 1: Create a new E2B sandbox
# Set timeout to at least 5 minutes (300 seconds) so sandbox stays alive
sandbox = Sandbox.create(timeout=600)  # 10 minutes to be safe

print(f"Created sandbox with ID: {sandbox.sandbox_id}")

# Step 2: Write index.html file in /home/user/ inside the sandbox
index_html_content = "Hello from E2B Sandbox!"
sandbox.files.write("/home/user/index.html", index_html_content)
print("Wrote index.html to /home/user/index.html")

# Step 3: Start a Python HTTP server in the background on port 8000 serving /home/user/
# Use the --directory flag to serve /home/user/
# Start the server with nohup in background, detached from this shell
start_cmd = "nohup python3 -m http.server 8000 --directory /home/user > /tmp/http_server.log 2>&1 & echo $! > /tmp/http_server.pid; disown; sleep 1"
sandbox.commands.run(start_cmd, background=True)
print("Started HTTP server on port 8000")

# Wait a moment for the server to start
time.sleep(5)

# Verify the server is running
result = sandbox.commands.run("ps aux | grep http.server | grep -v grep")
print(f"Server process: {result.stdout}")

result = sandbox.commands.run("curl -s http://localhost:8000/")
print(f"HTTP server response: {result.stdout}")

# Step 4: Retrieve the public URL for port 8000
host = sandbox.get_host(8000)
public_url = f"https://{host}"
print(f"Public URL: {public_url}")

# Step 5: Save the sandbox ID and public URL to /home/user/e2b_task_info.json on the local machine
info = {
    "sandbox_id": sandbox.sandbox_id,
    "url": public_url,
}

local_path = "/home/user/e2b_task_info.json"
with open(local_path, "w") as f:
    json.dump(info, f, indent=2)

print(f"Saved info to {local_path}")
print(json.dumps(info, indent=2))

# Step 6: Keep the sandbox alive for at least 5 minutes (300 seconds)
# We'll wait 310 seconds to ensure it's kept alive
keep_alive_seconds = 310
print(f"Keeping sandbox alive for {keep_alive_seconds} seconds...")
time.sleep(keep_alive_seconds)

print("Done!")
