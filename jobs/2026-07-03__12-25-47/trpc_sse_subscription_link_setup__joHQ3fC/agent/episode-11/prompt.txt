import { createTRPCClient } from '@trpc/client';
import { httpSubscriptionLink } from '@trpc/client';
import fs from 'fs';
import type { AppRouter } from './server';

const client = createTRPCClient<AppRouter>({
  links: [
    httpSubscriptionLink({
      url: 'http://localhost:3000',
    }),
  ],
});

async function main() {
  const results: number[] = [];
  const subscription = client.countdown.subscribe(3, {
    onData: (value: number) => {
      results.push(value);
    },
    onComplete: () => {
      fs.writeFileSync('/home/user/project/output.json', JSON.stringify(results));
      console.log('Wrote output:', results);
    },
    onError: (err: unknown) => {
      console.error('Subscription error:', err);
    },
  });

  // Wait for the subscription to complete
  await new Promise<void>((resolve) => {
    subscription.unsubscribe();
    // Give some time for the data to be flushed
    setTimeout(resolve, 2000);
  });
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
