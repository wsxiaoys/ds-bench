"""Build the barley grouped-bar chart HTML."""

import os
import altair as alt
from altair.datasets import data


def main() -> None:
    output_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), "output")
    os.makedirs(output_dir, exist_ok=True)

    barley_url = data.barley.url

    # Aggregated mean yield per (site, variety) for the grouped bars
    base_bars = (
        alt.Chart(barley_url)
        .mark_bar()
        .encode(
            x=alt.X(
                "mean(yield):Q",
                title="Mean Yield",
            ),
            y=alt.Y(
                "site:N",
                sort=alt.SortField(
                    field="yield",
                    op="mean",
                    order="descending",
                ),
                title="Site",
            ),
            color=alt.Color(
                "variety:N",
                scale=alt.Scale(scheme="tableau10"),
                title="Variety",
            ),
            xOffset=alt.XOffset("variety:N", title="Variety"),
            tooltip=["site:N", "variety:N", "mean(yield):Q"],
        )
        .properties(width=240, height=180)
    )

    # Per-site mean yield across all varieties (within each year facet)
    site_mean = (
        alt.Chart(barley_url)
        .transform_aggregate(
            mean_yield="mean(yield)",
            groupby=["site", "year"],
        )
        .mark_tick(color="black", thickness=3, size=40)
        .encode(
            x="mean_yield:Q",
            y="site:N",
        )
    )

    layered = alt.layer(base_bars, site_mean).facet(
        facet=alt.Facet("year:O", title="Year"),
        columns=2,
    )

    chart = layered.properties(
        title=alt.Title(
            text="Barley Yield by Variety and Site",
            subtitle="Grouped bars show mean yield per (site, variety); black tick marks show per-site mean across varieties.",
        )
    )

    out_path = os.path.join(output_dir, "chart.html")
    chart.save(out_path)
    print(f"Saved chart to {out_path}")


if __name__ == "__main__":
    main()
