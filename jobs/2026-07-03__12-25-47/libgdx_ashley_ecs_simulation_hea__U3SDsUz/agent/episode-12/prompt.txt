#!/usr/bin/env bash
# Entry point for the deterministic Ashley ECS simulation.
# Usage: run.sh <scenario-file>
#
# Only the simulation's stdout is propagated.  All Gradle progress output
# and any other chatter are diverted to stderr so they don't pollute the
# captured simulation result.

set -eu

HERE="$(cd "$(dirname "$0")" && pwd)"
cd "$HERE"

if [ "$#" -lt 1 ]; then
    echo "usage: run.sh <scenario-file>" >&2
    exit 2
fi

SCENARIO="$1"

# Run via `gradle -q` so Gradle swallows its own logging on stdout/stderr.
# -q still lets our program's System.out through (because System.out is
# inherited from the Java process Gradle launches).
#
# To make absolutely sure NO Gradle noise reaches stdout (some tasks can still
# print under -q), we explicitly redirect Gradle's 1/2 file descriptors and
# capture only the standard streams of the launched JVM.

gradle -q --console=plain run --args="\"$SCENARIO\"" '
' 2> >(grep -v -E '^(BUILD|FAILURE|> Configure|> Task|> :)' >&2 || true) > /tmp/gdx_ecs_stdout.tmp
status=$?

# If gradle itself errored, bubble that error up.
if [ $status -ne 0 ]; then
    cat /tmp/gdx_ecs_stdout.tmp >&2 || true
    rm -f /tmp/gdx_ecs_stdout.tmp
    exit "$status"
fi

cat /tmp/gdx_ecs_stdout.tmp
rm -f /tmp/gdx_ecs_stdout.tmp
exit 0
