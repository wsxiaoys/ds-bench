use std::collections::BTreeMap;

    dotenvy::dotenv().ok();
    let convex_url = env::var("CONVEX_URL")?;

    let args: Vec<String> = env::args().collect();
    if args.len() < 2 {
        eprintln!("Usage: rust-client <text>");
        std::process::exit(1);
    }
    let text = args[1].clone();

    let run_id = fs::read_to_string("/logs/artifacts/run-id")?;
    let run_id = run_id.trim().to_string();

    let mut client = ConvexClient::new(&convex_url)?;
    let mut map = BTreeMap::new();
    map.insert("text".to_string(), text);
    map.insert("runId".to_string(), run_id);
    client.mutation("tasks:create", map).await?;

    Ok(())
}
