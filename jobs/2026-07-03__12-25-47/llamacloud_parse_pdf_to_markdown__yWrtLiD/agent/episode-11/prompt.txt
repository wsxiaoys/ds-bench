import { LlamaCloud } from '@llamaindex/llama-cloud';
import * as fs from 'fs';
import * as path from 'path';

async function main(): Promise<void> {
  // Create output directory if it doesn't exist
  const outputDir = path.join(__dirname, 'output');
  if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true });
  }

  // Initialize the client. The SDK reads LLAMA_CLOUD_API_KEY from the environment REDACTEDmatically.
  const client = new LlamaCloud();

  // Upload the PDF file with the correct purpose for a parse job.
  const fileStream = fs.createReadStream(path.join(__dirname, 'sample.pdf'));
  const uploadedFile = await client.files.create({
    file: fileStream as any,
    purpose: 'parse',
  });
  const fileId = uploadedFile.id;
  console.log(`Uploaded file with ID: ${fileId}`);

  // Run a synchronous parse job (this blocks until the job is COMPLETED).
  const result = await client.parsing.parse({
    file_id: fileId,
    tier: 'cost_effective',
    version: 'latest',
    expand: ['markdown'],
  });

  const jobId = result.job.id;
  const jobStatus = result.job.status;
  console.log(`Parse job ID: ${jobId}, status: ${jobStatus}`);

  // Verify the job can be retrieved via client.parsing.get (sanity check).
  const retrieved = await client.parsing.get(jobId);
  console.log(`Retrieved job status: ${retrieved.job.status}`);

  // Collect markdown from every page.
  const pages = result.markdown?.pages ?? [];
  const pageMarkdownList: string[] = [];
  for (const page of pages) {
    if ('markdown' in page && page.success) {
      pageMarkdownList.push(page.markdown);
    } else {
      console.warn(`Page ${page.page_number} failed: ${('error' in page) ? page.error : 'unknown'}`);
    }
  }
  const pageCount = pageMarkdownList.length;

  // Concatenate pages with `---` between them.
  const combinedMarkdown = pageMarkdownList.join('\n---\n');

  // Persist the parsed markdown.
  const parsedMdPath = path.join(outputDir, 'parsed.md');
  fs.writeFileSync(parsedMdPath, combinedMarkdown, 'utf8');
  console.log(`Wrote parsed markdown to ${parsedMdPath}`);

  // Append a human-readable log entry.
  const logPath = path.join(outputDir, 'result.log');
  const logLines = [
    `File ID: ${fileId}`,
    `Job ID: ${jobId}`,
    `Job Status: COMPLETED`,
    `Page Count: ${pageCount}`,
  ];
  fs.appendFileSync(logPath, logLines.join('\n') + '\n', 'utf8');
  console.log(`Appended log to ${logPath}`);
}

main().catch((err) => {
  console.error('Error during parsing:', err);
  process.exit(1);
});
