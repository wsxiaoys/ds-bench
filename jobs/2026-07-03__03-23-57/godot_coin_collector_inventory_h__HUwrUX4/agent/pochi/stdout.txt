### Key Implementations

#### 1. Autoload Singleton (`REDACTEDloads/Inventory.gd`)
Manages the global coin count and persists it as JSON:
```gdscript
extends Node

signal coin_changed(new_count: int)

var coin_count: int = 0

func _ready() -> void:
    self.load()

func _notification(what: int) -> void:
    if what == NOTIFICATION_WM_CLOSE_REQUEST:
        save()

func _exit_tree() -> void:
    save()

func add_coin() -> void:
    coin_count += 1
    coin_changed.emit(coin_count)

func get_count() -> int:
    return coin_count

func save() -> void:
    var file = FileAccess.open("user://save.json", FileAccess.WRITE)
    if file:
        var data = {"count": coin_count}
        file.store_string(JSON.stringify(data))
        file.close()

func load() -> void:
    if not FileAccess.file_exists("user://save.json"):
        coin_changed.emit(coin_count)
        return
    
    var file = FileAccess.open("user://save.json", FileAccess.READ)
    if file:
        var json_string = file.get_as_text()
        file.close()
        var data = JSON.parse_string(json_string)
        if data is Dictionary and data.has("count"):
            coin_count = int(data["count"])
        coin_changed.emit(coin_count)
```

#### 2. Player Movement (`scenes/Player.gd`)
Handles standard arrow-key movement inside the physics process:
```gdscript
extends CharacterBody2D

const SPEED = 200.0

func _physics_process(_delta: float) -> void:
    var direction_x := Input.get_axis("ui_left", "ui_right")
    var direction_y := Input.get_axis("ui_up", "ui_down")
    
    velocity.x = direction_x * SPEED
    velocity.y = direction_y * SPEED
    
    move_and_slide()
```

#### 3. Coin Collection (`scenes/Coin.gd`)
Senses player entry, triggers inventory updates, and frees itself:
```gdscript
extends Area2D

signal collected

func _ready() -> void:
    body_entered.connect(_on_body_entered)

func _on_body_entered(body: Node2D) -> void:
    if body.name == "Player" or body is CharacterBody2D:
        collected.emit()
        Inventory.add_coin()
        queue_free()
```

#### 4. Reactive HUD Overlay (`scenes/HUD.gd`)
Updates the UI labels immediately when the `coin_changed` signal is emitted:
```gdscript
extends CanvasLayer

@onready var label: Label = $Label

func _ready() -> void:
    Inventory.coin_changed.connect(_on_coin_changed)
    _on_coin_changed(Inventory.get_count())

func _on_coin_changed(new_count: int) -> void:
    label.text = "Coins: " + str(new_count)
```
