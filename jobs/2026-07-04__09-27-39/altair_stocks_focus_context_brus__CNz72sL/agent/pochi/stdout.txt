from vega_datasets import data
import altair as alt

brush = alt.selection_interval(encodings=['x'])
base = alt.Chart(data.sp500.url).mark_area().encode(x='date:T', y='price:Q')
detail = base.encode(x=alt.X('date:T', scale=alt.Scale(domain=brush))).properties(width=600, height=250)
context = base.properties(width=600, height=70).add_params(brush)
chart = detail & context

# Get the spec
spec = chart.to_dict()
