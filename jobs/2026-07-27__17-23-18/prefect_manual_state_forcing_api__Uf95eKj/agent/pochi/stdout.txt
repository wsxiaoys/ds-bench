from prefect.states import Completed, Failed, Cancelled, Crashed

@flow(name='state-forcing-flow-zrlbk7b3p3')
def state_forcing_flow():
    pass

async def main():
    async with get_client() as client:
        # Define the runs and their desired final states
        runs_to_create = [
            ('ingest-zrlbk7b3p3', Completed()),
            ('transform-zrlbk7b3p3', Failed()),
            ('validate-zrlbk7b3p3', Cancelled()),
            ('publish-zrlbk7b3p3', Crashed())
        ]
        
        for name, state in runs_to_create:
            # Create a flow run
            flow_run = await client.create_flow_run(flow=state_forcing_flow, name=name)

            res = await client.set_flow_run_state(flow_run_id=flow_run.id, state=state, force=True)

            # Verify state

            print(f'Verified state of {name}: {updated.state.name}')

asyncio.run(main())
