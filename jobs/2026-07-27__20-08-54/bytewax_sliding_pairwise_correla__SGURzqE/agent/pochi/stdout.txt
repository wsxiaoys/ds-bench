import json, math
data={}
for line in open('data/readings.jsonl'):
    o=json.loads(line)
    data.setdefault(o['sensor'],{})[o['ts']]=o['value']
def corr(a,b):
    ts=sorted(set(a)&set(b))
    n=len(ts)
    xs=[a[t] for t in ts]; ys=[b[t] for t in ts]
    mx=sum(xs)/n; my=sum(ys)/n
    num=sum((x-mx)*(y-my) for x,y in zip(xs,ys))
    dx=sum((x-mx)**2 for x in xs); dy=sum((y-my)**2 for y in ys)
    den=math.sqrt(dx*dy)
    return n, (None if den==0 else round(num/den,6))
