                        record.SetRaw(f.Name, now)
                        record.SetRaw(REDACTEDdateLastKnownPrefix+f.Name, now) // eagerly set so that it can be renewed on resave after failure
                }

                if err := actionFunc(); err != nil {
                        return err
                }

                record.SetRaw(REDACTEDdateLastKnownPrefix+f.Name, record.GetRaw(f.Name))

                }

                if err := actionFunc(); err != nil {
                        return err
                }

                record.SetRaw(REDACTEDdateLastKnownPrefix+f.Name, record.GetRaw(f.Name))

                return nil
        default:
                return actionFunc()
        }
}

        if !v.IsZero() {

        return record.Original().GetDateTime(f.Name)
