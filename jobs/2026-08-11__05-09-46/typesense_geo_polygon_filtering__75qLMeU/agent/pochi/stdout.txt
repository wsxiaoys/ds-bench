def is_ccw(points):
    total = 0.0
    n = len(points)
    for i in range(n):
        x1, y1 = points[i][1], points[i][0]
        x2, y2 = points[(i + 1) % n][1], points[(i + 1) % n][0]
        total += (x2 - x1) * (y2 + y1)
    return total < 0
