import requests
import json

TYPESENSE_URL = 'http://localhost:8108'
HEADERS = {'X-TYPESENSE-API-KEY': 'xyz'}

# Load documents
raw_docs = []
with open('/home/user/import-pipeline/data/raw_products.jsonl', 'r') as f:
    for line in f:
        if line.strip():
            raw_docs.append(json.loads(line))

# Convert to JSONL
payload = '\n'.join(json.dumps(doc) for doc in raw_docs)

# Import
params = {
    'action': 'create',
    'dirty_values': 'coerce_or_reject'
}
res = requests.post(f'{TYPESENSE_URL}/collections/catalog/documents/import', headers=HEADERS, params=params, data=payload)
print('Status:', res.status_code)
# Print first few response lines and any failure lines
response_lines = res.text.strip().split('\n')
print('Total response lines:', len(response_lines))
for i, line in enumerate(response_lines):
    resp_obj = json.loads(line)
    if not resp_obj.get('success'):
        print(f'Line {i} failed: {resp_obj}')
