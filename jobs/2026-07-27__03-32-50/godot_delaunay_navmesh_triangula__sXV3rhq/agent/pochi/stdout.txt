extends SceneTree

func _cross(o, a, b):
	return (a.x-o.x)*(b.y-o.y) - (a.y-o.y)*(b.x-o.x)

func _on_segment(p,q,r):
	return min(p.x,r.x)-1e-7<=q.x and q.x<=max(p.x,r.x)+1e-7 and min(p.y,r.y)-1e-7<=q.y and q.y<=max(p.y,r.y)+1e-7

func _segments_intersect(p1,p2,p3,p4):
	var d1=_cross(p3,p4,p1)
	var d2=_cross(p3,p4,p2)
	var d3=_cross(p1,p2,p3)
	var d4=_cross(p1,p2,p4)
	var eps=1e-9
	if ((d1>eps and d2<-eps) or (d1<-eps and d2>eps)) and ((d3>eps and d4<-eps) or (d3<-eps and d4>eps)):
		return true
	if abs(d1)<=eps and _on_segment(p3,p1,p4): return true
	if abs(d2)<=eps and _on_segment(p3,p2,p4): return true
	if abs(d3)<=eps and _on_segment(p1,p3,p2): return true
	if abs(d4)<=eps and _on_segment(p1,p4,p2): return true
	return false

func _is_valid_bridge(combined, hole, i, j):
	var v = combined[i]
	var m = hole[j]
	var n = combined.size()
	for k in range(n):
		var a=combined[k]; var b=combined[(k+1)%n]
		if k==i or (k+1)%n==i: continue
		if _segments_intersect(v,m,a,b): return false
	var hn=hole.size()
	for k in range(hn):
		var a=hole[k]; var b=hole[(k+1)%hn]
		if k==j or (k+1)%hn==j: continue
		if _segments_intersect(v,m,a,b): return false
	return true
