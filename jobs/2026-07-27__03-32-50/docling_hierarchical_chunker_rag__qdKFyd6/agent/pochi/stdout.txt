from docling.document_converter import DocumentConverter
from docling_core.types.doc.document import SectionHeaderItem, TitleItem
from docling_core.transforms.chunker.hierarchical_chunker import HierarchicalChunker

conv = DocumentConverter()
res = conv.convert('assets/report.pdf')
doc = res.document

headers = [item for item, _ in doc.iterate_items(with_groups=True) if isinstance(item, (SectionHeaderItem, TitleItem))]
heights = sorted({round(h.prov[0].bbox.t - h.prov[0].bbox.b, 1) for h in headers}, reverse=True)
rank = {h: i+1 for i, h in enumerate(heights)}
for h in headers:
    if isinstance(h, SectionHeaderItem):
        ht = round(h.prov[0].bbox.t - h.prov[0].bbox.b, 1)
        h.level = rank[ht]
        print(h.text, '->level', h.level)

chunker = HierarchicalChunker()
chunks = list(chunker.chunk(doc))
for c in chunks:
    print('----')
    print(c.meta.headings)
    print(c.text[:80])
    print([di.prov[0].page_no for di in c.meta.doc_items if di.prov])
