import json
from datetime import datetime

with open("occupancy_proof.json") as f:
    proof = json.load(f)

assert set(proof.keys()) == {"limit_name", "total_slots", "tasks"}
assert proof["limit_name"] == "critical-section"
assert proof["total_slots"] == 4
tasks = proof["tasks"]
assert len(tasks) == 8

expected_slots = {"t0":1,"t1":1,"t2":2,"t3":1,"t4":2,"t5":1,"t6":2,"t7":1}
seen_ids = set()
events = []
for t in tasks:
    assert set(t.keys()) == {"task_id","slots","entered_at","exited_at"}
    tid = t["task_id"]
    assert tid in expected_slots and tid not in seen_ids
    seen_ids.add(tid)
    assert t["slots"] == expected_slots[tid]
    entered = datetime.fromisoformat(t["entered_at"])
    exited = datetime.fromisoformat(t["exited_at"])
    assert entered.tzinfo is not None and exited.tzinfo is not None
    assert entered < exited
    assert (exited - entered).total_seconds() >= 1.0
    events.append((entered, 1, t["slots"], tid))   # enter
    events.append((exited, 0, -t["slots"], tid))   # exit (process exits before enters at same time, hence 0 sort key)
assert seen_ids == set(expected_slots.keys())

events.sort(key=lambda e: (e[0], e[1]))
occ = 0
max_occ = 0
for ts, kind, delta, tid in events:
    occ += delta
    max_occ = max(max_occ, occ)
    assert occ <= 4, f"Exceeded capacity: {occ} at {ts} due to {tid}"
assert max_occ == 4, f"Never reached ceiling of 4, max was {max_occ}"
print("All checks passed. Max concurrent occupancy:", max_occ)
