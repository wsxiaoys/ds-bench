import json, math, re, sys, os

# Inline the BM25Index class to verify it loads from JSON in a fresh process
class BM25Index:
    def __init__(self, k1=1.5, b=0.75, epsilon=0.25):
        self.k1 = k1
        self.b = b
        self.epsilon = epsilon
        self.corpus_size = 0
        self.avgdl = 0.0
        self.doc_len = []
        self.doc_freqs = []
        self.idf = {}
        self.average_idf = 0.0
        self._built = False

    @classmethod
    def from_dict(cls, d):
        idx = cls(k1=d['k1'], b=d['b'], epsilon=d['epsilon'])
        idx.corpus_size = d['corpus_size']
        idx.avgdl = d['avgdl']
        idx.doc_len = d['doc_len']
        idx.doc_freqs = d['doc_freqs']
        idx.idf = d['idf']
        idx.average_idf = d['average_idf']
        idx._built = True
        return idx

    def get_scores(self, query_tokens):
        scores = np.zeros(self.corpus_size)
        doc_len_arr = np.array(self.doc_len)
        for q in query_tokens:
            q_freq = np.array([doc.get(q, 0) for doc in self.doc_freqs])
            idf_q = self.idf.get(q, 0.0)
            scores += idf_q * (
                q_freq * (self.k1 + 1)
                / (q_freq + self.k1 * (1 - self.b + self.b * doc_len_arr / self.avgdl))
            )
        return scores

    def top_k(self, query_tokens, k=5):
        scores = self.get_scores(query_tokens)
        if k >= self.corpus_size:
            top_indices = list(range(self.corpus_size))
        else:
            top_indices = np.argpartition(scores, -k)[-k:]
        top_indices = top_indices[np.argsort(-scores[top_indices])]
        results = []
        for idx in top_indices:
            if scores[idx] > 0:
                results.append({'chunk_id': int(idx), 'score': float(scores[idx])})
        return results[:k]

def tokenize(text):
    return [t for t in re.split(r'[^a-zA-Z0-9]+', text.lower()) if t]

## Key Design Decisions

- **Structural chunking**: One chunk per Docling structural element (no fixed-size windows)
- **Heading enrichment**: `heading_path` captures hierarchical scope; heading-only queries match heading chunks because headings are indexed as their own chunks
- **Table preservation**: Cell content extracted verbatim row-by-row with `|` separators
- **BM25**: ATIRE variant (matching `rank_bm25.BM25Okapi`), self-contained implementation with JSON persistence (no pickle dependency issues)
- **Fully offline**: No network calls, no embedding models, pure lexical retrieval with regex-based tokenization
